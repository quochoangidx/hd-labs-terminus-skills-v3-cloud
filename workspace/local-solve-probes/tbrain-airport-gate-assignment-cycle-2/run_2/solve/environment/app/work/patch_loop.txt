    if random.random() < LNSP:
        res = lns(t)
        if res is None: continue
        old, d = res
        if d <= 0 or random.random() < math.exp(-d / Tcur):
            cur += d
            if cur < best: best = cur; bestwhere = where[:]
        else:
            for x in old: occ[where[x]].discard(x)
            for x, s in old.items(): where[x] = s; occ[s].add(x)
        continue
